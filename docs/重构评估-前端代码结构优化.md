# 前端代码结构优化重构评估报告

## 执行摘要

本报告分析了前端代码的结构问题，发现 `HomeView.vue` 组件职责过重（1245行代码）和 `knowledge.ts` Store 职责不清的问题。通过引入 Composables 模式进行重构，可以将代码量减少 76%，显著提升代码的可维护性、可测试性和复用性。

# 1. 问题

前端代码存在严重的职责分散和耦合问题，违反了高内聚低耦合原则，主要体现在 `HomeView.vue` 组件职责过重和 `knowledge.ts` Store 职责不清。

5. **挂载模式管理职责** - 模式切换、锚点计算
6. **UI 状态管理职责** - 样式计算、状态管理
7. **测试数据生成职责** - 测试数据生成（开发工具）

## 3.1. **提取主题管理 Composable**

创建 `composables/useTheme.ts`，封装主题管理逻辑：

```typescript
import { ref, watch, onMounted } from 'vue';

export type Theme = 'black' | 'blue';

export function useTheme() {
  const isDark = ref(false);
  const currentTheme = ref<Theme>('blue');

  const applyTheme = (theme: Theme) => {
    if (theme === 'black') {
      document.documentElement.setAttribute('data-theme', 'black');
    } else {
      document.documentElement.removeAttribute('data-theme');
    }
  };

  const toggleTheme = () => {
    currentTheme.value = currentTheme.value === 'black' ? 'blue' : 'black';
    isDark.value = currentTheme.value === 'black';
  };

  const initTheme = () => {
    const savedTheme = localStorage.getItem('treeTheme') as Theme | null;
    if (savedTheme) {
      currentTheme.value = savedTheme;
      isDark.value = savedTheme === 'black';
    }
    applyTheme(currentTheme.value);
  };

  watch(currentTheme, (newTheme) => {
    applyTheme(newTheme);
    localStorage.setItem('treeTheme', newTheme);
  });

  onMounted(() => {
    initTheme();
  });

  return { isDark, currentTheme, toggleTheme };
}
```

**使用示例**：
```typescript

const { isDark, toggleTheme } = useTheme();
```

## 3.2. **提取聊天交互 Composable**

创建 `composables/useChat.ts`，封装聊天交互逻辑：

```typescript
import { ref, nextTick } from 'vue';
import { ElMessage } from 'element-plus';
import { useKnowledgeStore } from '../stores/knowledge';

export function useChat(chatListRef: Ref<HTMLElement | null>) {
  const store = useKnowledgeStore();
  const newQuestion = ref('');
  const pendingQuestion = ref('');

  const scrollToBottom = async () => {
    await nextTick();
    if (chatListRef.value) {
      chatListRef.value.scrollTop = chatListRef.value.scrollHeight;
    }
  };

  const handleAsk = async (options: { context?: string; parentId?: string }) => {
    if (!newQuestion.value.trim()) return;

    const question = newQuestion.value;
    pendingQuestion.value = question;
    newQuestion.value = '';

    try {
      await store.askQuestion(question, options.parentId, options.context);
      ElMessage.success('问题已回答！');
      await scrollToBottom();
    } catch (error: any) {
      console.error(error);
      if (error.code === 'ECONNABORTED') {
        ElMessage.error('请求超时，本地模型生成较慢，请耐心等待或检查 Ollama 状态');
      } else {
        ElMessage.error('获取回答失败，请检查网络或后端服务');
      }
    } finally {
      pendingQuestion.value = '';
    }
  };

  return { newQuestion, pendingQuestion, handleAsk, scrollToBottom };
}
```

```typescript

import { ElMessage } from 'element-plus';
import { useKnowledgeStore } from '../stores/knowledge';
import { exportKnowledgeAsXMind, exportKnowledgeAsMarkdownZip, triggerDownload } from '../utils/knowledgeExport';

export function useExport() {
  const store = useKnowledgeStore();
  const exportingXMind = ref(false);
  const exportingMarkdown = ref(false);
  const exportScope = ref<'all' | 'selected-path'>('all');
  const exportTargetNodeIds = ref<string[]>([]);

  const buildNodeChain = (nodeId: string) => {
    const chain: string[] = [];
    let cursor = store.nodes.find(n => n._id === nodeId);
    while (cursor) {
      chain.unshift(cursor._id);
      if (!cursor.parentId) break;
      cursor = store.nodes.find(n => n._id === cursor.parentId);
    }
    return chain;
  };

  const exportPathNodeIds = computed(() => {
    if (exportScope.value !== 'selected-path') return [];
    const pathSet = new Set<string>();
    exportTargetNodeIds.value.forEach(nodeId => {
      buildNodeChain(nodeId).forEach(id => pathSet.add(id));
    });
    return Array.from(pathSet);
  });

  const handleExportXMind = () => {
    const nodes = getExportNodes();
    if (!nodes || !nodes.length) {
      ElMessage.warning('暂无可导出的知识节点');
      return;
    }
    exportingXMind.value = true;
    try {
      const blob = exportKnowledgeAsXMind(nodes);
      triggerDownload(blob, `AI学习树-${getExportFileTime()}.xmind`);
      ElMessage.success('XMind 知识框架图已开始下载');
    } finally {
      exportingXMind.value = false;
    }
  };

  return {
    exportingXMind,
    exportingMarkdown,
    exportScope,
    exportPathNodeIds,
    handleExportXMind,
    handleExportMarkdown
  };
}
```

```typescript
import { ElMessage } from 'element-plus';

export interface ContextItem {
  id: string;
  text: string;
  sourceNodeId: string;
}

export function useContext() {
  const selectedContexts = ref<ContextItem[]>([]);
  const dragFromIndex = ref<number | null>(null);

    const trimmedText = text.trim();

      item => item.text === trimmedText && item.sourceNodeId === sourceNodeId

    if (!exists) {
      selectedContexts.value.push({
        id: `${sourceNodeId}-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
        text: trimmedText,
        sourceNodeId

        message: `已加入上下文（${selectedContexts.value.length}段）`,
        duration: 2000
      });
      return true;
    }
    return false;
  };

  const mergedContext = computed(() => {
    return selectedContexts.value.map(item => item.text).join('\n');
  });

  return {
    selectedContexts,
    addContext,
    removeContext,
    clearAllContexts,
    mergedContext
  };
}
```

## 3.5. **提取挂载模式 Composable**

创建 `composables/useMountMode.ts`，封装挂载模式逻辑：

```typescript
import { ref, computed } from 'vue';
import { useKnowledgeStore } from '../stores/knowledge';
import type { ContextItem } from './useContext';

export type MountMode = 'followup' | 'context-anchor';

export function useMountMode(selectedContexts: Ref<ContextItem[]>) {
  const store = useKnowledgeStore();
  const askMode = ref<MountMode>('followup');

  const buildNodeChain = (nodeId: string) => {
    const chain: string[] = [];
    let cursor = store.nodes.find(n => n._id === nodeId);
    while (cursor) {
      chain.unshift(cursor._id);
      if (!cursor.parentId) break;
      cursor = store.nodes.find(n => n._id === cursor.parentId);
    }
    return chain;
  };

  const findCommonAncestor = (nodeIds: string[]) => {
    if (nodeIds.length === 0) return undefined;
    if (nodeIds.length === 1) return nodeIds[0];

    const chains = nodeIds.map(id => buildNodeChain(id));
    let index = 0;
    let lastCommon = '';

    while (true) {
      const current = chains[0]?.[index];
      if (!current) break;
      if (chains.every(chain => chain[index] === current)) {
        lastCommon = current;
        index += 1;
        continue;
      }
      break;
    }

    return lastCommon || nodeIds[0];
  };

  const anchorTargetNodeId = computed(() => {
    if (askMode.value !== 'context-anchor') return null;
    const uniqueIds = [...new Set(selectedContexts.value.map(item => item.sourceNodeId))];
    return findCommonAncestor(uniqueIds) || null;
  });

  return { askMode, anchorTargetNodeId };
}
```

**使用示例**：
```typescript
import { useMountMode } from '../composables/useMountMode';
const { askMode, anchorTargetNodeId } = useMountMode(selectedContexts);
```

## 3.6. **提取树结构构建工具函数**

创建 `utils/treeBuilder.ts`，提取树结构构建逻辑：

```typescript
import type { KnowledgeNode } from '../types';

export interface TreeNode extends KnowledgeNode {
  children: TreeNode[];
  level: number;
}

export function buildTree(nodes: KnowledgeNode[], sessionId: string): TreeNode[] {
  if (!nodes.length) return [];

  const nodeMap = new Map<string, TreeNode>();
  const roots: TreeNode[] = [];

  nodes.forEach(node => {
    nodeMap.set(node._id, { ...node, children: [], level: 0 });
  });

  nodes.forEach(node => {
    const mappedNode = nodeMap.get(node._id);
    if (!mappedNode) return;

    if (node.parentId && nodeMap.has(node.parentId)) {
      const parent = nodeMap.get(node.parentId);
      if (parent) {
        parent.children.push(mappedNode);
      }
    } else {
      roots.push(mappedNode);
    }
  });

  // BFS 设置层级
  const queue = roots.map(root => ({ node: root, level: 0 }));
  while (queue.length > 0) {
    const item = queue.shift();
    if (!item) continue;
    const { node, level } = item;
    node.level = level;
    
    if (node.children && node.children.length > 0) {
      node.children.forEach((child) => {
        queue.push({ node: child, level: level + 1 });
      });
    }
  }

  // 多根节点时创建虚拟根
  if (roots.length > 1) {
    return [{
      _id: 'root',
      question: '知识根节点',
      answer: '这是所有知识的起点',
      sessionId,
      createdAt: new Date().toISOString(),
      level: -1,
      children: roots
    }];
  }

  return roots;
}
```

**使用示例**：
```typescript
import { buildTree } from '../utils/treeBuilder';
const treeData = computed(() => buildTree(nodes.value, sessionId.value));
```
